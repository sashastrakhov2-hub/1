git clone - клонирование 
git status - показать состояние рабочего каталога 
git commit - сохранить изменения из индекса в репозитории 
git push - отправить локальные коммиты в удалённый репозиторий 
git pull - получение из удалённого репозитория и слить с локальной веткой 
git fitch - скачать изменения из удалённого репозитория но не сливать 
git branch - управление ветками 
git checkout -  переключиться на другую ветку 
git merge - слить изменения из одной ветки в текущую 
git log - показать историю коммитов
